\chapter{تحلیل نیازمندی‌ها}

نیازمندی‌های این پروژه از دل تجربه کاربر بیرون می‌آیند. کاربر مهمان می‌خواهد سرویس‌ها را راحت پیدا کند؛ توسعه‌دهنده می‌خواهد بعد از ورود، حساب، مصرف و ابزارهای خود را مدیریت کند؛ مدیر هم باید بتواند سامانه را کنترل و نگهداری کند. بنابراین نیازمندی‌ها فقط یک فهرست اداری نیستند، بلکه توضیح می‌دهند هر بخش از سامانه باید چه مشکلی را حل کند.

\section{نقش‌های کاربری}
سه نقش اصلی در پروژه دیده می‌شود: مهمان، توسعه‌دهنده احرازشده و مدیر سیستم. مهمان با بخش عمومی فهرست سروکار دارد. توسعه‌دهنده پس از ورود، وارد فضای کاری خود می‌شود و با پیشخوان، انتشار رابط، ابزار فراخوان و محیط طراحی جریان کار می‌کند. مدیر سیستم هم مسئول کنترل کلی و نگهداری داده‌هاست.

مهمان بدون ورود می‌تواند بخش عمومی بازارچه را ببیند؛ این بخش برای معرفی رابط‌ها و ایجاد اعتماد اولیه ضروری است. توسعه‌دهنده پس از احراز هویت به امکاناتی می‌رسد که با داده خصوصی او ارتباط دارند، مانند کلید دسترسی، سازمان، اشتراک، مصرف، انتشار و ابزارهای آزمایشی. نقش مدیر سیستم بیشتر به نگهداری داده و پشتیبانی از مسیرهای مدیریتی مربوط می‌شود و به همین دلیل از نظر سطح دسترسی حساس‌تر است.

\subsection{نمودارهای موارد کاربرد}
برای جلوگیری از ابهام در نقش‌ها و مرز سامانه، هر مورد کاربرد با یک نمودار مستقل زبان مدل‌سازی پلنت‌یو‌ام‌ال\footnote{\lr{PlantUML}} در همان صفحه مشخصات نمایش داده شده است. بازیگر اصلی و بازیگران بیرونی خارج از مرز \lr{IranAPI} قرار دارند و رابطه‌های «شامل»\footnote{\lr{include}} و «گسترش»\footnote{\lr{extend}} فقط رفتارهای واقعاً پیاده‌سازی‌شده را نشان می‌دهند. فایل‌های منبع نمودارها همراه پروژه نگهداری می‌شوند تا نمودار و پیاده‌سازی هم‌زمان قابل اصلاح باشند.

\UseCaseStart
\subsection{مورد کاربرد \lr{UC-01}: کشف و جستجوی رابط}
\noindent\textbf{بازیگر اصلی:} مهمان یا توسعه‌دهنده احرازشده.\\
\textbf{هدف:} یافتن یک رابط فعال بر اساس مسئله، حوزه کاربرد یا محبوبیت.\\
\textbf{پیش‌شرط:} فهرست در دسترس باشد؛ ورود به حساب لازم نیست.\\
\textbf{محرک:} کاربر صفحه مرور سرویس‌ها را باز می‌کند یا عبارت جست‌وجو و پالایه را تغییر می‌دهد.

\UseCaseDiagram{uc-01}{\lr{UC-01}}{uc01}

\noindent\textbf{جریان اصلی:}
\begin{enumerate}
  \item سامانه فهرست دسته‌ها و رابط‌های فعال را از سمت سرور دریافت می‌کند.
  \item کاربر می‌تواند عبارت جستجو، دسته، برچسب، وضعیت ویژه یا محبوب و ترتیب نمایش را مشخص کند.
  \item سمت سرور شرط‌ها را روی فهرست اعمال و نتیجه را صفحه‌بندی می‌کند.
  \item رابط کاربری برای هر نتیجه، اطلاعات خلاصه مانند نام، توضیح کوتاه، دسته، امتیاز و قیمت پایه را نشان می‌دهد.
  \item کاربر یکی از نتایج را برای مطالعه بیشتر انتخاب می‌کند.
\end{enumerate}

\noindent\textbf{مسیرهای جایگزین و خطا:}
اگر هیچ سرویس منطبق نباشد، سامانه حالت خالی را نشان می‌دهد و کاربر می‌تواند پالایه‌ها را حذف کند. سرویس غیرفعال در نتیجه عمومی دیده نمی‌شود. درخواست مشاهده سرویس‌های متعلق به کاربر نیز فقط پس از احراز هویت پذیرفته می‌شود.\\
\textbf{پس‌شرط:} فهرست منطبق با معیارهای کاربر نمایش داده می‌شود؛ تا زمانی که جزئیات یک سرویس باز نشود، داده عملیاتی کاربر تغییر نمی‌کند.

\UseCaseStart
\subsection{مورد کاربرد \lr{UC-02}: بررسی جزئیات و مستندات}
\noindent\textbf{بازیگر اصلی:} مهمان یا توسعه‌دهنده احرازشده.\\
\textbf{هدف:} ارزیابی فنی و اقتصادی سرویس پیش از استفاده.\\
\textbf{پیش‌شرط:} رابط انتخاب‌شده فعال و دارای شناسه متنی معتبر باشد.\\
\textbf{محرک:} کاربر کارت یک سرویس یا پیوند مستقیم صفحه جزئیات را باز می‌کند.

\UseCaseDiagram{uc-02}{\lr{UC-02}}{uc02}

\noindent\textbf{جریان اصلی:}
\begin{enumerate}
  \item سامانه اطلاعات کامل سرویس را با دسته، توضیحات، نشانی پایه و روش احراز هویت دریافت می‌کند.
  \item طرح‌های قیمت‌گذاری، مستندات و نقاط پایانی ثبت‌شده کنار اطلاعات اصلی نمایش داده می‌شوند.
  \item کاربر روش، مسیر، ورودی و توضیح هر نقطه پایانی را بررسی می‌کند.
  \item سامانه سرویس‌های مشابه را بر اساس داده فهرست پیشنهاد می‌دهد.
  \item شمار مشاهده سرویس برای سنجش میزان توجه کاربران افزایش می‌یابد.
\end{enumerate}

\noindent\textbf{مسیرهای جایگزین و خطا:}
اگر سرویس وجود نداشته یا غیرفعال باشد، پاسخ «یافت نشد» برگردانده می‌شود. نبود طرح، مستندات یا نقطه پایانی مانع نمایش اطلاعات پایه نیست و رابط کاربری برای آن بخش حالت خالی نشان می‌دهد.\\
\textbf{پس‌شرط:} کاربر اطلاعات لازم برای مقایسه، انتخاب طرح، آزمایش یا کنارگذاشتن سرویس را در اختیار دارد.

\UseCaseStart
\subsection{مورد کاربرد \lr{UC-03}: ثبت‌نام، ورود و خروج}
\noindent\textbf{بازیگران اصلی:} مهمان برای ثبت‌نام و ورود؛ توسعه‌دهنده احرازشده برای خروج.\\
\textbf{هدف:} ایجاد حساب جدید، بازیابی نشست حساب موجود یا پایان امن نشست.\\
\textbf{پیش‌شرط:} برای ثبت‌نام، نام کاربری و رایانامه باید تکراری نباشند؛ برای ورود، حساب باید از قبل وجود داشته باشد.\\
\textbf{محرک:} کاربر فرم ثبت‌نام یا ورود را ارسال می‌کند، یا فرمان خروج را می‌زند.

\UseCaseDiagram{uc-03}{\lr{UC-03}}{uc03}

\noindent\textbf{جریان اصلی ثبت‌نام و ورود:}
\begin{enumerate}
  \item کاربر مشخصات هویتی و گذرواژه را وارد می‌کند.
  \item سامانه کامل‌بودن داده، تطابق گذرواژه‌ها و یکتایی مشخصات را اعتبارسنجی می‌کند.
  \item در ثبت‌نام، سند کاربر و نمایه ساخته می‌شود؛ در ورود، گذرواژه با داده ذخیره‌شده تطبیق داده می‌شود.
  \item سمت سرور یک نشست پایدار می‌سازد و شناسه آن را با کوکی \lr{HttpOnly} به مرورگر می‌دهد.
  \item اطلاعات عمومی حساب و نمایه برای راه‌اندازی فضای خصوصی به رابط کاربری برگردانده می‌شود.
\end{enumerate}

\noindent\textbf{جریان خروج:}
سامانه نشست سمت سرور را حذف، کوکی مرورگر را منقضی و وضعیت کاربر را مهمان می‌کند.\\
\textbf{مسیرهای جایگزین و خطا:}
نام کاربری یا رایانامه تکراری، عدم تطابق گذرواژه‌ها و اطلاعات ورود نادرست با خطای اعتبارسنجی پاسخ داده می‌شوند. ارائه‌دهنده ورود اجتماعی فقط در صورت پیکربندی و فعال‌بودن قابل استفاده است؛ ارائه‌دهنده ناشناخته یا غیرفعال کاربر را وارد سامانه نمی‌کند.\\

  \item تغییر مستقیماً در مجموعه عملیاتی مربوط ذخیره می‌شود تا نتیجه در فهرست یا بخش حساب قابل مشاهده باشد.
  \item مجموعه مصرف رابط برای مدیر سیستم فقط‌خواندنی است و امکان افزودن، ویرایش یا حذف ندارد.
\end{enumerate}

\noindent\textbf{مسیرهای جایگزین و خطا:}
کاربر عادی و کارمند بدون نقش توسعه‌دهنده رابط نمی‌توانند کنسول زنده را باز کنند. دسترسی توسعه‌دهنده به سند متعلق به کاربر دیگر رد می‌شود. داده نامعتبر ذخیره نمی‌شود و عملیات تغییر روی مجموعه فقط‌خواندنی پذیرفته نمی‌گردد.\\
\textbf{پس‌شرط:} داده معتبر در مجموعه زنده ذخیره شده و دامنه مالکیت و سطح دسترسی حفظ شده است.

\subsection{مورد کاربرد \lr{UC-13}: تحلیل آرشیو پروژه}
\noindent\textbf{بازیگر اصلی:} توسعه‌دهنده احرازشده.\\
\textbf{هدف:} شناخت ساختار یک پروژه فشرده پیش از اجرای آن و دریافت برآورد آمادگی استقرار.\\
\textbf{پیش‌شرط:} نشست معتبر وجود داشته باشد و فایل \lr{ZIP}، \lr{TAR}، \lr{TAR.GZ} یا \lr{TGZ} حداکثر ۲۵ مگابایت باشد.\\
\textbf{محرک:} کاربر آرشیو پروژه را در ابزار تحلیل بارگذاری می‌کند.

\noindent\textbf{جریان اصلی:}
\begin{enumerate}

  \item محتوای متنی بدون اجرای کد بررسی می‌شود تا زبان از میان شش زمان‌اجرای مدیریت‌شده تشخیص داده شود.
  \item چارچوب، فایل‌های ورودی، فایل‌های مانیفست، \lr{Dockerfile}، مسیرهای \lr{HTTP} و پورت احتمالی استخراج می‌شوند.
  \item هش آرشیو و هشدارهایی مانند نبود نقطه ورود، مسیرهای پویا یا وجود فایل رازدار در نتیجه ثبت می‌شود.
  \item سامانه سازنده پیشنهادی، فرمان شروع، زمان‌اجرای مقصد و مقدار آمادگی استقرار را برمی‌گرداند.
\end{enumerate}

\noindent\textbf{مسیرهای جایگزین و خطا:}
آرشیو خالی، خراب، بزرگ‌تر از حد، دارای مسیر پیمایش، پیوند نمادین یا پسوند پشتیبانی‌نشده با خطای اعتبارسنجی رد می‌شود. زبان ناشناخته یا نبود مسیر \lr{HTTP} هشدار تولید می‌کند، اما در صورت وجود \lr{Dockerfile} می‌تواند همچنان قابل استقرار باشد.\\
\textbf{پس‌شرط:} نتیجه تحلیل بدون اجرای کد در اختیار کاربر است و فایل خام برای این مسیر ذخیره نمی‌شود.

\subsection{مورد کاربرد \lr{UC-14}: استقرار پروژه فشرده}
\noindent\textbf{بازیگر اصلی:} توسعه‌دهنده احرازشده.\\
\textbf{هدف:} پذیرش، ساخت و اجرای یک پروژه رابط در محیط جداشده و قابل پایش.\\
\textbf{پیش‌شرط:} آرشیو تحلیل‌شده باید زمان‌اجرای پشتیبانی‌شده، نقطه ورود یا \lr{Dockerfile} داشته باشد؛ نام پروژه و ناحیه نیز معتبر باشند.\\
\textbf{محرک:} کاربر آرشیو، نام پروژه و یکی از نواحی مجاز را برای استقرار ارسال می‌کند.

\noindent\textbf{جریان اصلی:}
\begin{enumerate}

  \item یک رکورد استقرار با وضعیت \lr{queued} ساخته می‌شود و کارگر استقرار آن را به وضعیت \lr{building} می‌برد.
  \item کارگر آرشیو را با کنترل مسیر استخراج می‌کند و از \lr{Dockerfile} موجود یا سازنده زبان‌محور استفاده می‌کند.

  \item در شکست ساخت یا خروج برنامه، وضعیت \lr{failed} و دلیل خلاصه ذخیره می‌شود و کاربر می‌تواند جزئیات استقرار را ببیند.
\end{enumerate}

\noindent\textbf{مسیرهای جایگزین و خطا:}
نام کوتاه یا طولانی، ناحیه نامعتبر، پروژه آماده‌نشده، آرشیو خارج از فضای اختصاصی، نبود \lr{Docker} یا پایان زمان ساخت باعث رد یا شکست کنترل‌شده می‌شود. استقرار هر کاربر فقط با شناسه همان کاربر قابل مشاهده است.\\
\textbf{پس‌شرط:} یا استقرار با وضعیت \lr{deployed} و نشانی قابل بررسی ثبت شده است، یا وضعیت \lr{failed} همراه دلیل قابل پیگیری باقی می‌ماند.

\textbf{پس‌شرط:} نتیجه عملیات در قالب انسانی یا \lr{JSON} قابل پردازش در اختیار اسکریپت یا توسعه‌دهنده قرار می‌گیرد.

\section{ورودی‌ها، خروجی‌ها و اعتبارسنجی}
ورودی‌های اصلی شامل مشخصات حساب، نمایه، شناسه یا نامک رابط، پالایه‌های جست‌وجو، امتیاز، داده انتشار رابط، شناسه طرح اشتراک، گره‌های محیط طراحی، بایگانی فشرده پروژه، نام پروژه و ناحیه است. در معماری مبتنی بر \lr{Django REST Framework}، معمولاً تبدیل‌گر داده همین داده‌ها را کنترل، پاک‌سازی و به شکل قابل استفاده تبدیل می‌کند \cite{drf}.

خروجی‌های اصلی سامانه در قالب \lr{JSON} بازگردانده می‌شوند و برای مصرف سمت کاربر طراحی شده‌اند. پاسخ‌های عمومی معمولاً داده فهرست، مستندات، طرح‌ها و وضعیت سلامت را شامل می‌شوند. پاسخ‌های خصوصی، اطلاعاتی مانند نمایه، سازمان‌ها، کلید پنهان‌شده، اشتراک و تاریخچه مصرف را برمی‌گردانند. جداسازی این دو دسته مهم است، چون داده عمومی برای همه کاربران قابل نمایش است، اما داده خصوصی باید فقط بعد از احراز هویت برگردد.

اعتبارسنجی در این پروژه دو کار اصلی انجام می‌دهد: جلوی ذخیره داده ناقص یا نامعتبر را می‌گیرد و خطا را به پاسخی قابل فهم برای کارخواه تبدیل می‌کند. برای نمونه، هنگام ثبت امتیاز باید مقدار امتیاز معتبر باشد، هنگام انتشار رابط فیلدهای اصلی باید حاضر باشند و هنگام اجرای محیط طراحی ساختار گره‌ها باید قابل پردازش باشد.

\section{امنیت}
احراز هویت باید طوری انجام شود که کاربر بعد از ورود به بخش‌های خصوصی برسد، اما داده حساس او در پاسخ‌ها آشکار نشود. در پروژه‌ای که با کلید رابط، توکن و کوکی کار می‌کند، این موضوع فقط یک جزئیات جانبی نیست؛ بخشی از امنیت اصلی محصول است \cite{django,drf}.

از دید نیازمندی، امنیت فقط ورود کاربر نیست. کنترل دسترسی، نگهداری نشست، جلوگیری از افشای رازها و محدودکردن خروجی‌های حساس هم بخشی از همین موضوع‌اند. آزمون‌های موجود نشان می‌دهند که پنهان‌سازی داده حساس در خروجی‌ها بررسی شده است. تصویر تولیدی رازها را از محیط می‌خواند، حالت اشکال‌زدایی را غیرفعال می‌کند و کوکی امن، \lr{HSTS}، \lr{CSP} و شناخت میانجی \lr{HTTPS} را اعمال می‌کند؛ گام باقی‌مانده ثبت راز ثابت چرخش‌پذیر، محدودسازی دقیق میزبان‌ها و پایش خطاهای عملیاتی است.

\section{جمع‌بندی فصل}
نیازهای پروژه حول کشف عمومی رابط، فضای کاری توسعه‌دهنده، مدیریت حساب و اشتراک، و امنیت نشست شکل گرفته‌اند. همین نیازها مبنای فصل معماری هستند؛ چون لایه‌های سامانه باید بتوانند نقش‌ها، جریان‌ها و محدودیت‌های امنیتی را پشتیبانی کنند.
